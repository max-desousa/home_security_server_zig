// Global state
let currentPath = "";
let hasVideos = false;

function isVideoFile(name) {
  const lower = name.toLowerCase();
  return lower.endsWith(".ts") || lower.endsWith(".mp4");
}

function addRow(name, url, isdir, size, size_string, date_modified, date_modified_string) {
  if (name === "." || name === "..") return;

  var root = document.location.pathname;
  if (root.substr(-1) !== "/") root += "/";

  var tbody = document.getElementById("tbody");
  var row = document.createElement("tr");

  if (isdir) {
    name += "/";
    url += "/";
    size = 0;
    size_string = "";

    var file_cell = document.createElement("td");
    var link = document.createElement("a");
    link.className = "icon dir";
    link.innerText = name;
    link.href = root + url;

    file_cell.dataset.value = name;
    file_cell.appendChild(link);

    row.appendChild(file_cell);
    row.appendChild(createCell(size, size_string));
    row.appendChild(createCell(date_modified, date_modified_string));
  } else {
    if (isVideoFile(name)) hasVideos = true;

    var file_cell = document.createElement("td");
    var link = document.createElement("a");
    link.className = "icon file";
    link.innerText = name;
    link.href = root + url;

    file_cell.dataset.value = name;
    file_cell.appendChild(link);

    var actions_cell = document.createElement("td");
    actions_cell.style.whiteSpace = "nowrap";

    var viewBtn = document.createElement("a");
    viewBtn.href = root + url;
    viewBtn.className = "button";
    viewBtn.innerText = "▶ View";
    viewBtn.style.marginRight = "8px";
    actions_cell.appendChild(viewBtn);

    var dlBtn = document.createElement("a");
    dlBtn.href = root + url;
    dlBtn.className = "button danger";
    dlBtn.innerText = "↓ Download";
    dlBtn.download = name;
    actions_cell.appendChild(dlBtn);

    row.appendChild(file_cell);
    row.appendChild(actions_cell);
    row.appendChild(createCell(size, size_string));
    row.appendChild(createCell(date_modified, date_modified_string));
  }

  tbody.appendChild(row);
}

function createCell(value, text) {
  var cell = document.createElement("td");
  cell.className = "detailsColumn";
  cell.dataset.value = value;
  cell.innerText = text;
  return cell;
}

function start(location) {
  currentPath = location;
  document.getElementById("header").innerText = 
    document.getElementById("header").innerText.replace("LOCATION", location);
  document.getElementById("title").innerText = document.getElementById("header").innerText;
}

function onHasParentDirectory() {
  const box = document.getElementById("parentDirLinkBox");
  box.style.display = "block";
  const root = document.location.pathname.endsWith("/") ? 
               document.location.pathname : document.location.pathname + "/";
  document.getElementById("parentDirLink").href = root + "..";
}

// ====================== NEW: CLEAN YYYYMMDD_HHMM ======================
function formatToFilePrefix(datetimeLocal) {
  // datetimeLocal = "2026-04-18T10:02"
  return datetimeLocal
    .replace(/-/g, '')      // remove all dashes → 20260418T1002
    .replace('T', '_')      // → 20260418_1002
    .replace(':', '');      // → 20260418_1002
}

function generateM3U8Playlist() {
  const startInput = document.getElementById("startTime").value;
  const endInput   = document.getElementById("endTime").value;

  if (!startInput || !endInput) {
    alert("Please select both Start and End time.");
    return;
  }

  const cleanStart = formatToFilePrefix(startInput);
  const cleanEnd   = formatToFilePrefix(endInput);

  let url = currentPath;
  if (!url.endsWith("/")) url += "/";

  // This is the important change:
  url += "?m3u8&start=" + cleanStart + "&end=" + cleanEnd + "&player=true";

  console.log("📡 Opening Player with playlist:", url);
  window.open(url, "_blank");
}

function initializeVideoPlaylistUI() {
  if (hasVideos) {
    document.getElementById("videoPlaylistPanel").style.display = "block";
  }
}

function finalizeDirectory() {
  initializeVideoPlaylistUI();
}
